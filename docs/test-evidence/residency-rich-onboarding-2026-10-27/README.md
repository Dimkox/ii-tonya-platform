Run this from the repository root to verify the package without running any witness:

```python
from pathlib import Path
import gzip
import hashlib
import json

root = Path("docs/test-evidence/residency-rich-onboarding-2026-10-27")
sha = lambda value: hashlib.sha256(value).hexdigest()
index = json.loads(Path(
    "docs/test-evidence/residency-rich-onboarding-implementation-2026-10-27.json"
).read_text())
assert sha((root / "manifest.json").read_bytes()) == index["durableArchiveManifest"]["sha256"]
manifest = json.loads((root / "manifest.json").read_text())
for entry in manifest["files"]:
    packed = (root / entry["file"]).read_bytes()
    raw = gzip.decompress(packed)
    assert len(packed) == entry["bytes"]
    assert sha(packed) == entry["sha256"]
    assert len(raw) == entry["rawBytes"]
    assert sha(raw) == entry["rawSha256"]
print("Verified", len(manifest["files"]), "archives")
```
